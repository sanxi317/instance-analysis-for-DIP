\begin{frame}
\frametitle{预处理：统一肺窗和灰度范围}
\textbf{HU：CT值的单位；肺窗突出肺部组织的灰度差异。}\par
\vspace{2mm}
窗位 $L=-600$ HU，窗宽 $W=1500$ HU：
\[
 a=L-\frac{W}{2}=-1350,\qquad b=L+\frac{W}{2}=150
\]
截断后，线性映射到 $0\sim255$：
\[
 I=255\,\frac{\operatorname{clip}(H,a,b)-a}{b-a}
\]
{\small $H$ 是原始 HU 值；$I$ 是显示与处理用的灰度值。}\par
\vspace{3mm}
$-1350$ HU $\rightarrow 0$\qquad $-600$ HU $\rightarrow$ 约 $128$\qquad $150$ HU $\rightarrow255$\par
\vspace{3mm}
\key{LDCT与FDCT使用相同窗口，评价才具有可比性。}
\source{教材对应：第3章灰度变换；SimpleITK读取获得HU值，随后截断、归一化和整数化。}
\note{预处理先把CT值转换到统一肺窗。窗位是显示区间的中心，窗宽是区间长度，因此上下限是负1350和150 HU。超出区间的值截断，再映射到0到255。这样输入和参考使用同一把灰度尺，避免显示差异影响评价。}
\end{frame}

% Slide 4
\begin{frame}
\frametitle{PSNR与SSIM：同时比较误差和结构}
{\small\textbf{PSNR：峰值信噪比}（Peak Signal-to-Noise Ratio）\par}
\[
 \mathrm{MSE}=\frac1N\sum_{k=1}^{N}(x_k-y_k)^2,\qquad
 \mathrm{PSNR}=10\log_{10}\frac{255^2}{\mathrm{MSE}}
\]
{\small $x$：参考；$y$：结果；$N$：像素数；MSE：均方误差。\par
\key{PSNR越高，通常像素误差越小。}}\par
\vspace{1mm}
{\small\textbf{SSIM：结构相似性指数}（Structural Similarity Index）\par}
\[
 \mathrm{SSIM}(x,y)=
 \frac{2\mu_x\mu_y+C_1}{\mu_x^2+\mu_y^2+C_1}
 \cdot
 \frac{2\sigma_{xy}+C_2}{\sigma_x^2+\sigma_y^2+C_2}
\]
{\small $\mu$：局部均值；$\sigma^2$：局部方差；$\sigma_{xy}$：局部协方差。\par
$C_1,C_2$ 防止分母过小；局部窗口计算后取平均。\par
\key{SSIM越接近1，亮度、对比度和结构越相似。}}\par
\source{输入须同尺寸、对应位置、同灰度范围；本次峰值255。PSNR单位dB；SSIM：Wang等，2004。}
\note{PSNR全称是Peak Signal-to-Noise Ratio，中文是峰值信噪比。先算对应像素的均方误差，再转换成对数指标，255是灰度峰值，误差小则分数高。SSIM全称是Structural Similarity Index，中文是结构相似性指数。它用局部均值、方差和协方差比较亮度、对比度与结构，再取平均。比较时尺寸、位置和灰度范围要一致；两项指标结合图像观察，不能直接当诊断准确率。}
\end{frame}

\begin{frame}
\frametitle{五种去噪方法的区别}
{\small
\renewcommand{\arraystretch}{1.38}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.18\textwidth}>{\raggedright\arraybackslash}p{0.47\textwidth}>{\raggedright\arraybackslash}p{0.29\textwidth}@{}}
\toprule
方法 & 核心做法 & 本次选择的参数 \\
\midrule
均值 & 邻域像素等权平均 & $3\times3$ \\
高斯 & 距离中心越近，平均权重越大 & $3\times3,\ \sigma=1$ \\
自适应中值 & 排序取中值，按条件扩大窗口 & 最大窗口 $5\times5$ \\
\textbf{非局部均值（NLM）} & \textbf{按图像块相似度加权平均} & $h=4$，块7，搜索窗21 \\
小波软阈值 & 小波分解，收缩细节系数，再重建 & db1，2层，阈值4 \\
\bottomrule
\end{tabular}\par}
\[
\mathbb E[(\widehat f_i-f_i)^2]=
\underbrace{(\mathbb E\widehat f_i-f_i)^2}_{\text{偏差平方}}
+\underbrace{\operatorname{Var}(\widehat f_i)}_{\text{随机波动}}
\]
\source{此为固定真实灰度$f$下的统计期望误差；平滑可降低波动，却增大边缘偏差。具体加权误差公式见后续页面。}
\note{五种方法怎样取舍，可用误差等于偏差平方加方差解释。均值、高斯平均能减少随机波动，但跨边缘平均会带来偏差；中值擅长抵抗脉冲异常值，小波阈值也可能删掉弱细节；NLM按相似块选权重，力求兼顾噪声和结构。本例好坏还要看实测指标。}
\end{frame}

% Slide 6
\begin{frame}
\frametitle{均值与高斯的权重}
{\small 简化为固定权重、独立零均值同方差噪声，$Y_j=f_j+\varepsilon_j$。}\par
\[
\widehat f_i=\sum_j w_jY_j,\qquad
\operatorname{MSE}=(\textstyle\sum_jw_jf_j-f_i)^2+\sigma_n^2\sum_jw_j^2
\]
均值：$w_j=1/k^2$，噪声方差降为$\sigma_n^2/k^2$。\par
若跨边界的总权重为$\alpha$、两侧灰度差为$\Delta$，偏差为$\alpha\Delta$。
\[
w_{\mathrm{Gaussian}}(s,t)=\frac1Z\exp\!\left[-\frac{s^2+t^2}{2\sigma_g^2}\right]
\]
{\small 高斯按距离减小远处权重，可能减轻结构混合；$\sigma_g$过大仍会模糊。\par
$Z$使权重和为1，$s,t$是空间偏移。}\par
\vspace{2mm}
\key{若边缘偏差增加超过噪声方差减少，MSE反而增大。}
\source{教材第3.4、3.5节；用偏差--方差机制解释本例，不把模型当作实测噪声估计。}
\note{把像素看成真实灰度加噪声。固定权重且噪声独立时，误差分为结构混合带来的偏差平方和加权噪声方差。均值的k乘k窗口能把方差降为原来的1/k平方，但跨边缘权重为α、两侧灰度差为Δ时，会产生αΔ的偏差。高斯降低远处权重，可能少混合一些结构；窗口或σ过大仍会模糊。这能解释为什么更平滑不一定带来更高PSNR。}
\end{frame}

% Slide 7
\begin{frame}
\frametitle{自适应中值的两阶段判断}
$z$：中心像素；$z_{\min},z_{\max},z_{\mathrm{med}}$：窗口的最小值、最大值和中值。
\vspace{3mm}
\textbf{阶段A：}是否满足 $z_{\min}<z_{\mathrm{med}}<z_{\max}$？
\begin{itemize}
\item 满足：进入阶段B。
\item 不满足：扩大窗口；达到最大窗口时输出中值。
\end{itemize}
\vspace{2mm}
\textbf{阶段B：}是否满足 $z_{\min}<z<z_{\max}$？
\begin{itemize}
\item 满足：保留中心像素 $z$。
\item 不满足：用 $z_{\mathrm{med}}$ 替换。
\end{itemize}
\source{教材第5.3节；本次窗口3×3起步，最大5×5。典型针对脉冲噪声，不等于对CT噪声总是最优。}
\note{自适应中值先判断窗口中值是否处在最小值和最大值之间；若不是，就扩大窗口。中值正常后，再判断中心像素是否异常，正常则保留，否则替换为中值。这个过程典型针对少数脉冲异常值，本例CT噪声不是已验证的纯脉冲模型，因此不能预期它一定最优。}
\end{frame}

% Slide 8
\begin{frame}
\frametitle{小波软阈值}
\textbf{小波分解 $\rightarrow$ 保留近似系数、收缩细节系数 $\rightarrow$ 逆变换}\par
\vspace{3mm}
对每个细节系数 $c$：
\[
T_\lambda(c)=\operatorname{sgn}(c)\max(|c|-\lambda,0)
\]
\begin{itemize}
\item $|c|\le\lambda$：变为0；$|c|>\lambda$：幅度减少 $\lambda$。
\item $\lambda=4$ 时：$3\rightarrow0$，$8\rightarrow4$，$-8\rightarrow-4$。
\item 本例：db1小波、2层分解，低频近似系数不做阈值处理。
\end{itemize}
\vspace{2mm}
\key{细节子带既有噪声，也有边缘；阈值过大会损伤结构。}
\source{教材第6.10节；软阈值来源：Donoho，IEEE TIT 1995。阈值作用于灰度域小波系数，不是HU值。}
\note{小波把图像分成低频近似与不同尺度的细节。我们保留近似系数，只对细节使用软阈值。绝对值不超过λ的系数变零，大于λ的则保留符号并收缩λ。举例λ等于4，3变成0，8变成4。噪声和真实边缘都可能出现在细节子带，所以阈值过大会损失有用结构。}
\end{frame}

% Slide 9
\begin{frame}
\frametitle{中值与软阈值为何有取舍}
\textbf{中值：最小化绝对偏差，不是平方误差}
\[
\operatorname{median}\{Y_j\}\in\arg\min_a\sum_j|Y_j-a|
\]
{\small 这是窗口中值的排序基础；自适应中值还需前述两阶段条件判断。\par
少数极端像素对中值影响小；若噪声不是少数极端值，\par
它不保证MSE最小，较大窗口也可能丢失细结构。}\par
\vspace{3mm}
\textbf{软阈值：拟合与稀疏惩罚的折中}
\[
\widehat c=\arg\min_z\left[\frac12(z-c)^2+\lambda|z|\right]
=\operatorname{sgn}(c)(|c|-\lambda)_+
\]
{\small $c$为观察到的小波细节系数，$\lambda\ge0$，$(u)_+=\max(u,0)$。\par
小系数置零能抑制噪声，但弱真实细节也可能被删；大系数收缩$\lambda$也带来偏差。}\par
\source{中值与小波阈值的数学机制；自适应窗口逻辑见前述判断流程，本例实际只对细节子带软阈值。}
\note{中值是绝对偏差之和的最小化解，所以少数极端值不容易把它拉走；但我们的PSNR基于平方误差，两种目标不等价。软阈值则是平方拟合误差加L1稀疏惩罚的最小化解。λ越大，系数越稀疏，但弱真实细节也越容易被删除，大系数收缩还会增加偏差。这些公式解释方法的取舍，不保证某种方法对本图指标最优。}
\end{frame}

% Slide 10
\begin{frame}
\frametitle{NLM：相似图像块贡献更大}
\textbf{比较一块邻域，而不只比较两个像素的灰度。}
\[
 \widehat I(i)=\sum_{j\in\Omega_i}w_{ij}I(j),\qquad
 w_{ij}=\frac{\exp(-d_{ij}^{\,2}/h^2)}{\displaystyle\sum_{k\in\Omega_i}\exp(-d_{ik}^{\,2}/h^2)}
\]
\begin{itemize}
\item $i$：待处理像素；$\Omega_i$：搜索窗口；$j$：候选像素。
\item $d_{ij}^{\,2}$：两块图像的灰度平方差距离；越小，权重越大。
\item $h$：控制相似度权重衰减；权重归一化后总和为1。
\end{itemize}
{\small\textbf{关键调用（本次选定参数）：}}\par
{\ttfamily\fontsize{9}{11}\selectfont
\noindent\detokenize{nlm = cv2.fastNlMeansDenoising(}\\
\hspace*{1.5em}\detokenize{ldct, None, h=4,}\\
\hspace*{1.5em}\detokenize{templateWindowSize=7, searchWindowSize=21)}\par}
\source{原理简化式：Buades等，CVPR 2005；调用取自任务二，ldct为统一肺窗uint8灰度图。}
\note{NLM按图像块相似度决定权重，块距离越小，权重越大，归一化后权重总和为1。下面是本次选中参数的调用：h等于4，图像块7乘7，搜索窗21乘21。它在有限窗口内寻找结构相似的区域，减少不同结构之间的平均，同时抑制随机波动。}
\end{frame}

% Slide 11
\begin{frame}
\frametitle{NLM的偏差与方差解释}
{\small 简化模型：$Y_j=f_j+\varepsilon_j$；$f_j$为真实灰度，$\varepsilon_j$为噪声。\par
假设权重固定，噪声独立、零均值、同方差$\sigma_n^2$，且$\sum_jw_j=1$。\par
估计式为$\widehat f_i=\sum_jw_jY_j$。}\par
\[
\operatorname{MSE}(\widehat f_i)=
\underbrace{\left[\sum_jw_j(f_j-f_i)\right]^2}_{\text{偏差平方}}
+\underbrace{\sigma_n^2\sum_jw_j^2}_{\text{噪声方差}}
\]
\begin{itemize}
\item 相似结构使$f_j$接近$f_i$，有助于减小偏差。
\item 若$m$个相似像素等权平均，噪声方差变为$\sigma_n^2/m$。
\item $h$过大趋向混合结构；过小则可参与平均的相似块太少。
\end{itemize}
{\small 实际NLM权重依赖带噪图像，CT噪声也未必符合这些假设；\par
此式用于解释机制，不是本实验的精确误差公式或最优性证明。}
\source{由加权估计的偏差--方差分解推导；NLM依据：Buades、Coll、Morel。}
\note{进一步看NLM的偏差和方差。假设权重固定、噪声独立且同方差，加权误差包含两项。相似结构使真实灰度接近，减小偏差；多个相似像素参与平均又降低噪声方差。如果h过大，简化指数权重趋向均匀，可能混入不同结构；h过小，又可能只剩自己或少量块，去噪变弱。实际NLM权重来自带噪图像，因此固定权重公式是机制模型，不能当本次精确误差分解。}
\end{frame}

\begin{frame}
\frametitle{双线性插值与四倍放大}
在相邻四点组成的单位网格中，$a,b\in[0,1]$为横纵相对距离：
\[
\begin{aligned}
I(a,b)={}&(1-a)(1-b)I_{00}+a(1-b)I_{10}\\
        &+(1-a)bI_{01}+abI_{11}
\end{aligned}
\]
{\small $I_{00},I_{10},I_{01},I_{11}$为四角像素值；四个权重非负且和为1；结果位于四角灰度极值之间。}\par
\vspace{3mm}
最近邻使用最近一点；双三次用$4\times4$邻域的三次核加权。\par
\vspace{3mm}
\key{宽、高各乘4 $\Rightarrow$ 像素总数乘16。}\par
跨边缘加权会把灰度突变变为渐变，可能模糊；更多采样不等于更多采集信息。
\source{教材第2.4节“图像插值”、第4.5节“重采样与插值”；源坐标到目标坐标的映射由OpenCV实现。}
\note{双线性用四个邻近像素的加权和生成中间值。a和b分别是横纵方向相对距离，四个权重非负且和为1，所以结果不会超过四角的灰度极值。跨边缘的凸组合会把突变变成渐变，可能模糊。每个方向放大4倍，像素总数乘16，但没有增加真实采集信息。}
\end{frame}

\begin{frame}
\frametitle{插值核与锯齿、模糊、过冲}
{\small 用一维形式说明；二维可沿两个方向使用核乘积。}\par
\[
\widehat f(x)=\sum_j I[j]K(x-j),\qquad
K_{\mathrm{NN}}(t)=\mathbf1_{[-1/2,\,1/2)}(t),\quad
K_{\mathrm{linear}}(t)=(1-|t|)_+
\]
{\small 最近邻为分段常数，易有台阶；线性核连续，但局部斜率会跳变。}\par
\[
K_a(t)=\begin{cases}
(a+2)|t|^3-(a+3)|t|^2+1,&|t|\le1,\\
a|t|^3-5a|t|^2+8a|t|-4a,&1<|t|<2,\\
0,&|t|\ge2.
\end{cases}
\]
{\small 三次核可使过渡更平滑，但含负权重。以$a=-0.75$、半格位置为例：\par
四点权重为$(-3,19,19,-3)/32$；若值为$(0,1,1,1)$，结果为$35/32>1$。\par
因此可能保留较强边缘，也可能过冲；更多点与更高阶不保证指标更好。}\par
\source{此为浮点三次核的说明性例子，整数输出还会饱和截断；不作理想高阶误差率或真细节恢复保证。}
\note{插值可以统一写成采样值与插值核的加权和。最近邻的矩形核给出分段常数，所以台阶和锯齿明显。线性三角核让灰度连续，但会平均两侧灰度。三次核有更平滑的局部过渡，却含负权重。半格位置的例子对0、1、1、1算出35/32，大于邻域最大值1，证明可能出现过冲。实际整数图像还会饱和截断，高阶核不保证每种指标都最好。}
\end{frame}

\begin{frame}
\frametitle{处理顺序为何影响结果}
{\small 将浮点插值简化为矩阵$U$；输入噪声独立、零均值，协方差为$\sigma_n^2 I_N$。}\par
\[
n'=Un,\qquad \operatorname{Cov}(n')=\sigma_n^2 UU^{\mathsf T}
\]
{\small 相邻输出共享输入像素，噪声通常相关；方差也受插值权重影响。}\par
\vspace{2mm}
{\small 把NLM简写为$D(y)=W(y)y$，其中$W(y)$是依赖带噪图像的权重矩阵：}\par
\[
U D(y)=U W(y)y,\qquad D(Uy)=W(Uy)Uy
\]
\[
D(Uy)-U D(y)=\bigl[W(Uy)U-UW(y)\bigr]y
\]
{\small 该差通常不为零：插值改变块相似度、噪声统计及权重。\par
放大4倍后，相同7/21像素窗口对应的原图物理范围约缩小到1/4。}\par
\vspace{2mm}
\key{解释“顺序有影响”，不证明“先去噪总是更好”。}
\source{此为机制模型；实际uint8还有舍入、饱和，CT噪声未必独立。本次只比较固定参数的两条流程。}
\note{最后从数学上解释处理顺序。把浮点插值写成矩阵U，独立输入噪声的协方差经插值变成σ平方乘UU转置。输出共享输入像素，因此噪声一般相关。NLM的权重W又依赖带噪图像，先插值再计算的权重不等于原尺寸权重。因此两种复合算子的差通常不为零。另外相同像素窗口在放大后对应更小的物理范围。这能解释顺序有影响，但不能证明先去噪总是最好。}
\end{frame}
